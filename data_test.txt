Что важно знать про это решение:

Один клик согласует все три типа за текущий период. Переключать rb больше не нужно.
Логика UpdateVSE не меняется: она по-прежнему берёт только записи, уже согласованные ответственным (o29620tvetstvenniiState == 1) и ещё не согласованные начальником (o296NachState != 1). То есть «Согласовать все» добивает именно начальницкий этап — это корректно.
Предусловие: все три списка (..._LS) должны быть загружены в памяти на момент клика. Судя по форме (три грида видны одновременно) — так и есть. Если у вас при выборе rb грузится только один список, добавьте в начало checkBox7_Click загрузку всех трёх (вызовы SetDataSourceOplataPrepod()/SetDataSourceWorkPOplata()/SetDataSourceStudentOplata()).
Кнопку всё равно надо разрешить для роли начальника (это отдельный гейт из п.1).
Если хочется оставить и «по отдельности» — этот вариант не мешает: при выбранной rb старое поведение можно сохранить на кнопке «Согласовать документ» (checkBox8), а «Согласовать все» сделать «по всем трём».

private void checkBox7_Click(object sender, EventArgs e)
{

    {

        SetDataSourceOplataPrepod();    // -> JournalDocumentOplataPrepod_LS  (Журнал)

        var journalIds = JournalDocumentOplataPrepod_LS

        var studentIds = AgreementOplataDocumentStudent_LS
            .Select(t => StaticFunction.GetIntForObject(t.Id, 0)).ToList();

    }

}

private void LoadVedomostView()
{
    int god = (int)numYearVed.Value;
    int mes = cmbMonthVed.SelectedIndex + 1;   // Январь=1 ... Декабрь=12
    if (mes <= 0) { MessageBox.Show("Выберите месяц."); return; }

    // Тот же набор, что уходит в перекачку: согласовано начальником за период.
    // Если три типа лежат в РАЗНЫХ таблицах/вью — замените на UNION ALL трёх SELECT.
    DataTable dt = Conn.Exec("year", $@"
        SELECT
            ROW_NUMBER() OVER (ORDER BY cex, tab_nom) AS [П/П],
            cex                       AS [Отдел],
            NSB.dbo.fn_FIO(tab_nom)   AS [Ф.И.О.],
            ''                        AS [Должность/профессия],  -- подставьте источник
            kateg                     AS [Кат.],
            tab_nom                   AS [Таб. №],
            hours                     AS [Часы],
            summ                      AS [Сумма доплаты],
            nom_dok                   AS [Документ]
        FROM Zarplata.dbo.Prepodavateli
        WHERE god = {god} AND mes = {mes} AND o2962_nach_State = 1
        ORDER BY cex, tab_nom");

    // Строка «Итого»
    if (dt.Rows.Count > 0)
    {
        decimal itogoChasy = dt.AsEnumerable().Sum(r => Convert.ToDecimal(r["Часы"]));
        decimal itogoSumma = dt.AsEnumerable().Sum(r => Convert.ToDecimal(r["Сумма доплаты"]));
        DataRow total = dt.NewRow();
        total["Ф.И.О."] = "Итого:";
        total["Часы"] = itogoChasy;
        total["Сумма доплаты"] = itogoSumma;
        dt.Rows.Add(total);
    }

    dgvVedomost.AutoGenerateColumns = true;
    dgvVedomost.DataSource = dt;   // напрямую, без BindingSource
}


SELECT ROW_NUMBER() OVER (ORDER BY v.ТипДок, v.[Отдел], v.[Таб. №]) AS [П/П], v.*
FROM (
    -- 1) Журнал (источник SetDataSourceOplataPrepod)
    SELECT 'Журнал' AS ТипДок,
           j.cex AS [Отдел], NSB.dbo.fn_FIO(j.tab_nom) AS [Ф.И.О.], '' AS [Должность/профессия],
           j.kateg AS [Кат.], j.tab_nom AS [Таб. №], j.hours AS [Часы], j.summ AS [Сумма доплаты], j.nom_dok AS [Документ]
    FROM <ТаблицаЖурнала> j
    WHERE j.god=@god AND j.mes=@mes AND j.o2962_nach_State=1

    UNION ALL
    -- 2) Соглашения с произв. рабочими (SetDataSourceWorkPOplata)
    SELECT 'Соглашение (раб.)',
           w.cex, NSB.dbo.fn_FIO(w.tab_nom), '', w.kateg, w.tab_nom, w.hours, w.summ, w.nom_dok
    FROM <ТаблицаСоглашенийРаб> w
    WHERE w.god=@god AND w.mes=@mes AND w.o2962_nach_State=1

    UNION ALL
    -- 3) Соглашения на практику студентов (SetDataSourceStudentOplata)
    SELECT 'Соглашение (студ.)',
           s.cex, NSB.dbo.fn_FIO(s.tab_nom), '', s.kateg, s.tab_nom,
           s.days /* у студентов часов нет — дни или 0 */, s.summ, s.nom_dok
    FROM <ТаблицаСоглашенийСтуд> s
    WHERE s.god=@god AND s.mes=@mes AND s.o2962_nach_State=1
) v
ORDER BY v.ТипДок, v.[Отдел], v.[Таб. №];
