2. Источник данных: BuildVedomostSource()

private List<BLVedomostWork> BuildVedomostSource()
{
    int year   = StaticFunction.GetIntForObject(cmbVedomostYear.Text, 0);
    int month  = cmbVedomostMonth.SelectedIndex + 1;
    bool allU  = true;   // вкладка-просмотр: по всем участкам (своего cmbUchastok тут нет)

    // "Согласованные" = подписка с o296NachState == 1
    var zak = ListData.GetSubscriptionOplataDocument().FindAll(s => s.o296NachState == 1);

    // --- ЖУРНАЛ: Vse -> Zakrit ---
    var journalZak =
        (from t in ListData.GetJournalDocumentOplataPrepod()
                          .Where(t => t.Year == year && t.Month == month)
         join s in zak on t.Id equals s.Id_OplataJornal
         select t).ToList();

    // --- СОГЛАШЕНИЯ (рабочие 3/4 + студенты 1/2): Vse -> Zakrit ---
    var agreementZak =
        (from t in ListData.GetAgreementDocumentOplataPrepod()
                          .Where(t => t.Year == year && t.Month == month)
         join s in zak on t.Id equals s.Id_OplataAgreement
         select t).ToList();

    var result = new List<BLVedomostWork>();

    // ЖУРНАЛ
    foreach (var t in journalZak)
        result.Add(new BLVedomostWork
        {
            Department = StaticFunction.GetStringForObject(t.InstructorDepartment),
            FIO        = StaticFunction.GetStringForObject(t.InstructorFIO),
            Post       = "",                                   // в журнале должности нет
            Category   = StaticFunction.GetStringForObject(t.CodeCathegory),
            TabNumber  = StaticFunction.GetStringForObject(t.Id_InstructorFIO),   // CHECK: таб.№ журнала
            Hours      = StaticFunction.GetIntForObject(t.CountWorkHours, 0),
            SumDoplata = StaticFunction.GetDecimalForObject(t.SumOplata, 0),
            Document   = StaticFunction.GetStringForObject(t.Number)
        });

    // СОГЛАШЕНИЯ С РАБОЧИМИ (TypeAgreement 3/4): Часы = CountWorkHours
    foreach (var t in agreementZak.Where(t => t.TypeAgreement == 3 || t.TypeAgreement == 4))
        result.Add(new BLVedomostWork
        {
            Department = StaticFunction.GetStringForObject(t.InstructorDepartment),
            FIO        = StaticFunction.GetStringForObject(t.InstructorFIO),
            Post       = StaticFunction.GetStringForObject(t.InstructorPost),
            Category   = StaticFunction.GetStringForObject(t.CodeCathegory),
            TabNumber  = StaticFunction.GetStringForObject(t.InstructorTabNumber),
            Hours      = StaticFunction.GetIntForObject(t.CountWorkHours, 0),       // рабочие → часы
            SumDoplata = StaticFunction.GetDecimalForObject(t.SumOpl, 0),
            Document   = StaticFunction.GetStringForObject(t.Number)
        });

    // СОГЛАШЕНИЯ НА ПРАКТИКУ СТУДЕНТОВ (TypeAgreement 1/2): Часы = Instructor_CountWorkDay (дни инструктора)
    foreach (var t in agreementZak.Where(t => t.TypeAgreement == 1 || t.TypeAgreement == 2))
        result.Add(new BLVedomostWork
        {
            Department = StaticFunction.GetStringForObject(t.InstructorDepartment),
            FIO        = StaticFunction.GetStringForObject(t.InstructorFIO),
            Post       = StaticFunction.GetStringForObject(t.InstructorPost),
            Category   = StaticFunction.GetStringForObject(t.CodeCathegory),
            TabNumber  = StaticFunction.GetStringForObject(t.InstructorTabNumber),
            Hours      = StaticFunction.GetIntForObject(t.Instructor_CountWorkDay, 0),  // студенты → дни инструктора
            SumDoplata = StaticFunction.GetDecimalForObject(t.SumOpl, 0),
            Document   = StaticFunction.GetStringForObject(t.Number)
        });

    int n = 1;
    foreach (var r in result) r.NN = n++;
    return result;
}

3. UI: новый tabPage + грид

private DataGridView dtgViewVedomost;
private TabPage _tabVedomost;        // ссылка нужна, чтобы узнавать переход на вкладку
private ComboBox cmbVedomostYear;    // СВОЙ выбор года на вкладке
private ComboBox cmbVedomostMonth;   // СВОЙ выбор месяца на вкладке

private void InitVedomostTab()
{
    _tabVedomost = new TabPage("Ведомость");

    // --- Панель фильтра (свои Год/Месяц, т.к. cmbYearDocument/cmbMonthDocument тут нет) ---
    var pnlFilter = new FlowLayoutPanel { Dock = DockStyle.Top, Height = 36, Padding = new Padding(6, 6, 0, 0) };

    pnlFilter.Controls.Add(new Label { Text = "Год:", AutoSize = true, Margin = new Padding(0, 8, 4, 0) });
    cmbVedomostYear = new ComboBox { DropDownStyle = ComboBoxStyle.DropDownList, Width = 80 };
    int curYear = DateTime.Now.Year;
    for (int y = curYear - 5; y <= curYear + 1; y++) cmbVedomostYear.Items.Add(y);
    cmbVedomostYear.SelectedItem = curYear;
    pnlFilter.Controls.Add(cmbVedomostYear);

    pnlFilter.Controls.Add(new Label { Text = "Месяц:", AutoSize = true, Margin = new Padding(12, 8, 4, 0) });
    cmbVedomostMonth = new ComboBox { DropDownStyle = ComboBoxStyle.DropDownList, Width = 120 };
    cmbVedomostMonth.Items.AddRange(new object[] {
        "Январь","Февраль","Март","Апрель","Май","Июнь",
        "Июль","Август","Сентябрь","Октябрь","Ноябрь","Декабрь" });
    cmbVedomostMonth.SelectedIndex = DateTime.Now.Month - 1;   // SelectedIndex + 1 = номер месяца
    pnlFilter.Controls.Add(cmbVedomostMonth);

    // --- Грид ---
    dtgViewVedomost = new DataGridView
    {
        Dock = DockStyle.Fill,
        AutoGenerateColumns = false,
        ReadOnly = true,
        AllowUserToAddRows = false,
        RowHeadersVisible = false,
        AllowUserToResizeColumns = true
    };
    AddVedomostColumns(dtgViewVedomost);

    // Порядок добавления важен: грид (Fill) добавляем ПЕРВЫМ, панель (Top) — ПОСЛЕ,
    // иначе панель перекроет верх грида.
    _tabVedomost.Controls.Add(dtgViewVedomost);
    _tabVedomost.Controls.Add(pnlFilter);
    tabControlMain.TabPages.Add(_tabVedomost);    // CHECK: имя вашего TabControl

    // Корректировка 2: автозагрузка вместо кнопки
    tabControlMain.SelectedIndexChanged += (s, e) =>
    {
        if (tabControlMain.SelectedTab == _tabVedomost)
            RefreshVedomost();
    };
    cmbVedomostYear.SelectedIndexChanged += VedomostFilterChanged;
    cmbVedomostMonth.SelectedIndexChanged += VedomostFilterChanged;
}

// Перестраиваем ведомость при смене года/месяца, но только если она сейчас открыта
private void VedomostFilterChanged(object sender, EventArgs e)
{
    if (_tabVedomost != null && tabControlMain.SelectedTab == _tabVedomost)
        RefreshVedomost();
}

private void AddVedomostColumns(DataGridView g)
{
    // Корректировка 3: ширины задаём в "весах" через FillWeight + режим Fill,
    // тогда колонки занимают всю ширину грида без серого хвоста справа.
    void Col(string prop, string header, int fillWeight, string fmt = null)
    {
        var c = new DataGridViewTextBoxColumn
        {
            DataPropertyName = prop,
            HeaderText = header,
            AutoSizeMode = DataGridViewAutoSizeColumnMode.Fill,
            FillWeight = fillWeight
        };
        if (fmt != null)
        {
            c.DefaultCellStyle.Format = fmt;
            c.DefaultCellStyle.Alignment = DataGridViewContentAlignment.MiddleRight;
        }
        g.Columns.Add(c);
    }

    Col("NN", "П/П", 5);
    Col("Department", "Отдел", 15);
    Col("FIO", "Ф.И.О.", 20);
    Col("Post", "Должность/профессия", 17);
    Col("Category", "Кат.", 5);
    Col("TabNumber", "Таб. №", 8);
    Col("Hours", "Часы", 6);
    Col("SumDoplata", "Сумма доплаты", 12, "N2");
    Col("Document", "Документ", 13);
}

private void RefreshVedomost()
{
    dtgViewVedomost.DataSource = BuildVedomostSource();
}

2) Процедура перекачки

CREATE PROCEDURE Zarplata.dbo.p_Перекачка_зп_преподавателей
    @god int, @mes int, @cex varchar(4)=NULL, @user varchar(50)=NULL, @tab int=NULL
AS
BEGIN
  SET NOCOUNT ON;
  UPDATE Zarplata.dbo.Prepodavateli
     SET o_svod_State=1, o_svod_User=ISNULL(@user,SUSER_SNAME()),
         o_svod_Date=GETDATE(), o_svod_Tab=@tab
   WHERE god=@god AND mes=@mes AND (@cex IS NULL OR cex=@cex)
     AND o2962_nach_State=1 AND o_svod_State=0;
  EXEC Zarplata.dbo.p_Экспорт_зп_преподавателей @god, @mes; -- реквизитный состав
END
